cmake_minimum_required(VERSION 3.20)
project(VoxlLogon LANGUAGES CXX)
add_library(voxl-logon SHARED bridge.cpp exports.def)
target_compile_features(voxl-logon PRIVATE cxx_std_17)
target_compile_definitions(voxl-logon PRIVATE UNICODE _UNICODE)
target_compile_options(voxl-logon PRIVATE /W4 /guard:cf)
target_link_options(voxl-logon PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
set_property(TARGET voxl-logon PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_executable(voxl-logon-test test.cpp)
target_compile_features(voxl-logon-test PRIVATE cxx_std_17)
target_compile_definitions(voxl-logon-test PRIVATE UNICODE _UNICODE)
target_compile_options(voxl-logon-test PRIVATE /UNDEBUG)
add_test(NAME credential-provider-contract COMMAND voxl-logon-test)
